<script>
  (function () {
    var root = document.getElementById('av-rv-{{ section.id }}');
    if (!root) return;
    var track = root.querySelector('[data-track]');
    var dotsBox = root.querySelector('[data-dots]');
    var prev = root.querySelector('[data-dir="-1"]'), next = root.querySelector('[data-dir="1"]');
    var auto = {{ section.settings.autoplay | json }}, timer = null;
    function cards() { return track.querySelectorAll('.av-rev'); }
    function step() { var c = cards(); if (c.length < 2) return track.clientWidth; return c[1].offsetLeft - c[0].offsetLeft; }
    function pages() { var c = cards().length, s = step(); var vis = Math.max(1, Math.round(track.clientWidth / s)); return Math.max(1, c - vis + 1); }
    function current() { return Math.round(track.scrollLeft / step()); }
    function update() {
      var n = pages(), i = Math.min(current(), n - 1);
      if (dotsBox.children.length !== n) {
        dotsBox.innerHTML = '';
        for (var k = 0; k < n; k++) {
          var d = document.createElement('button'); d.type = 'button'; d.className = 'av-rv__dot'; d.setAttribute('aria-label', 'Ir a la reseña ' + (k + 1));
          (function (k) { d.addEventListener('click', function () { go(k); }); })(k);
          dotsBox.appendChild(d);
        }
      }
      dotsBox.style.display = n > 1 ? '' : 'none';
      Array.prototype.forEach.call(dotsBox.children, function (d, k) { d.classList.toggle('is-on', k === i); });
      if (prev) prev.disabled = i <= 0;
      if (next) next.disabled = i >= n - 1;
    }
    function go(i) { var n = pages(); if (i < 0) i = 0; if (i > n - 1) i = n - 1; track.scrollTo({ left: i * step(), behavior: 'smooth' }); }
    root.querySelectorAll('.av-rv__btn').forEach(function (b) { b.addEventListener('click', function () { go(current() + Number(b.dataset.dir)); restart(); }); });
    var t; track.addEventListener('scroll', function () { clearTimeout(t); t = setTimeout(update, 80); }, { passive: true });
    window.addEventListener('resize', update);
    function restart() {
      if (!auto) return;
      clearInterval(timer);
      timer = setInterval(function () { var n = pages(), i = current(); go(i >= n - 1 ? 0 : i + 1); }, 5000);
    }
    track.addEventListener('pointerdown', function () { clearInterval(timer); });
    update(); restart();
  })();
</script>
