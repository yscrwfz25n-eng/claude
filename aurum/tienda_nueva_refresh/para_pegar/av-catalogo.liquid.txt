<script>
  (function () {
    var root = document.getElementById('av-cat-{{ sid }}');
    if (!root) return;
    function show(id) {
      root.querySelectorAll('.av-cat__tab').forEach(function (t) { var on = t.dataset.tab === id; t.classList.toggle('is-on', on); t.setAttribute('aria-selected', on ? 'true' : 'false'); });
      root.querySelectorAll('.av-cat__panel').forEach(function (p) { p.classList.toggle('is-on', p.dataset.panel === id); });
    }
    root.querySelectorAll('.av-cat__tab').forEach(function (t) {
      t.addEventListener('click', function () { show(t.dataset.tab); });
    });
    root.querySelectorAll('.av-cat__gal').forEach(function (g) {
      var tr = g.querySelector('[data-track]'), dots = g.querySelectorAll('.av-cat__dots i');
      function idx() { return Math.round(tr.scrollLeft / (tr.clientWidth || 1)); }
      function upd() { var i = idx(); dots.forEach(function (d, k) { d.classList.toggle('is-on', k === i); }); }
      g.querySelectorAll('.av-cat__arr').forEach(function (b) {
        b.addEventListener('click', function () {
          var n = tr.children.length, i = idx() + Number(b.dataset.dir);
          if (i < 0) i = n - 1; if (i > n - 1) i = 0;
          tr.scrollTo({ left: i * tr.clientWidth, behavior: 'smooth' });
        });
      });
      var t; tr.addEventListener('scroll', function () { clearTimeout(t); t = setTimeout(upd, 60); }, { passive: true });
    });
    document.addEventListener('shopify:block:select', function (e) {
      if (root.contains(e.target) && e.target.dataset && e.target.dataset.tab) show(e.target.dataset.tab);
    });
  })();
</script>
