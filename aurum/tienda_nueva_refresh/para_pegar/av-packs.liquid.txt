<script>
  (function () {
    var root = document.getElementById('av-pk-{{ section.id }}');
    if (!root) return;
    var bp = root.querySelector('[data-bp]');
    var radios = root.querySelectorAll('input[name="id"]');
    if (!root.querySelector('input[name="id"]:checked')) {
      var firstOk = root.querySelector('input[name="id"]:not(:disabled)');
      if (firstOk) firstOk.checked = true;
    }
    function upd() { var c = root.querySelector('input[name="id"]:checked'); if (c && bp) bp.textContent = c.dataset.price; }
    radios.forEach(function (r) { r.addEventListener('change', upd); });
    upd();
  })();
</script>
