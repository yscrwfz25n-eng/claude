<script>
  (function () {
    var root = document.querySelector('#MainContent .product__info-container');
    var src = document.getElementById('av-src');
    if (!root || !src) return;
    function top(el) { while (el && el.parentElement !== root) el = el.parentElement; return el; }
    function after(ref, el) { if (ref && el) ref.insertAdjacentElement('afterend', el); }
    var title = top(root.querySelector('.product__title'));
    var desc = top(root.querySelector('.product__description'));
    var picker = top(root.querySelector('variant-selects'));
    var buy = top(root.querySelector('product-form'));
    var packs = src.querySelector('[data-av-packs]');
    var perks = src.querySelector('[data-av-perks]');
    var faq = src.querySelector('[data-av-faq]');
    var bar = document.getElementById('AvSticky');
    if (title && desc) after(title, desc);
    after(picker || buy, packs);
    after(buy, perks);
    after(perks, faq);
    document.body.appendChild(bar);

    function radios() { return root.querySelectorAll('variant-selects input[type=radio]'); }
    function setPrice(t) { if (t) bar.querySelector('[data-av-price]').textContent = t; }
    if (radios().length) {
      root.classList.add('av-pk-ready');
      var checked = root.querySelector('variant-selects input[type=radio]:checked');
      packs.querySelectorAll('.av-pk__card').forEach(function (c) {
        var on = checked && c.dataset.value === checked.value;
        c.classList.toggle('is-on', !!on);
        if (on) setPrice(c.dataset.price);
      });
      packs.addEventListener('click', function (e) {
        var card = e.target.closest('.av-pk__card');
        if (!card || card.disabled) return;
        var r = Array.prototype.find.call(radios(), function (x) { return x.value === card.dataset.value; });
        if (r && !r.checked) r.click();
        packs.querySelectorAll('.av-pk__card').forEach(function (c) { c.classList.toggle('is-on', c === card); });
        setPrice(card.dataset.price);
      });
    } else { packs.remove(); }

    var target = root.querySelector('.product-form__submit');
    if (target && 'IntersectionObserver' in window) {
      new IntersectionObserver(function (en) {
        var show = !en[0].isIntersecting && en[0].boundingClientRect.top < 0;
        bar.classList.toggle('is-show', show);
        bar.setAttribute('aria-hidden', show ? 'false' : 'true');
        document.body.classList.toggle('av-sticky-on', show);
      }).observe(target);
      bar.querySelector('.av-sticky__btn').addEventListener('click', function () {
        var b = root.querySelector('.product-form__submit');
        if (b && !b.disabled) b.click();
      });
    }
  })();
</script>
