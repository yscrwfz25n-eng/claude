<script>
  (function () {
    var root = document.querySelector('#MainContent .product__info-container');
    var src = document.getElementById('av-src');
    if (!root || !src) return;
    document.body.classList.add('av-faja');
    function top(el) { while (el && el.parentElement !== root) el = el.parentElement; return el; }
    function after(ref, el) { if (ref && el) ref.insertAdjacentElement('afterend', el); }
    var title = top(root.querySelector('.product__title'));
    var buy = top(root.querySelector('product-form'));
    var bul = src.querySelector('[data-av-bul]');
    var wrap = src.querySelector('[data-av-packs-wrap]');
    var packs = src.querySelector('[data-av-packs]');
    var afterBuy = src.querySelector('[data-av-after]');
    after(title, bul);
    if (buy) buy.insertAdjacentElement('beforebegin', wrap);
    after(buy, afterBuy);

    var btn = root.querySelector('.product-form__submit');
    var btnPrice = null;
    if (btn) { btnPrice = document.createElement('span'); btnPrice.className = 'av-btn-price'; var first = btn.querySelector('span'); if (first) first.insertAdjacentElement('afterend', btnPrice); }
    function setBtn(card) { if (btnPrice && card) btnPrice.textContent = card.dataset.price; }

    function radios() { return root.querySelectorAll('variant-selects input[type=radio]'); }
    function pick(value) {
      var r = Array.prototype.find.call(radios(), function (x) { return x.value === value; });
      if (r && !r.checked) r.click();
      packs.querySelectorAll('.av-pk__card').forEach(function (k) { var on = k.dataset.value === value; k.classList.toggle('is-on', on); if (on) setBtn(k); });
    }
    if (!radios().length) { wrap.remove(); return; }
    root.classList.add('av-pk-ready');
    var checked = root.querySelector('variant-selects input[type=radio]:checked');
    if (checked) pick(checked.value);
    packs.addEventListener('click', function (e) {
      var card = e.target.closest('.av-pk__card');
      if (card && !card.disabled) pick(card.dataset.value);
    });

    /* 2 sobres elegido de entrada (salvo que vengan de un link con otro pack) */
    if (!/[?&]variant=/.test(location.search) && window.customElements) {
      customElements.whenDefined('variant-selects').then(function () {
        var rec = packs.querySelector('.av-pk__card[data-n="2"]:not([disabled])');
        if (rec) setTimeout(function () { pick(rec.dataset.value); }, 60);
      });
    }
  })();
</script>

<script>
  (function () {
    var CD = document.querySelector('cart-drawer');
    if (!CD) return;
    var ICON = {
      truck: '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linejoin="round"><path d="M3 6h11v9H3zM14 9h4l3 3v3h-7z"/><circle cx="7" cy="17" r="2"/><circle cx="17" cy="17" r="2"/></svg>',
      lock: '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><rect x="4" y="10" width="16" height="11" rx="2"/><path d="M8 10V7a4 4 0 0 1 8 0v3"/></svg>',
      shield: '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8"><path d="M12 3l8 3v6c0 5-3.5 8-8 9-4.5-1-8-4-8-9V6z"/><path d="M8.5 12l2.5 2.5L16 9.5"/></svg>',
      card: '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8"><rect x="3" y="6" width="18" height="13" rx="2"/><path d="M3 10h18M7 15h4"/></svg>'
    };
    function money(c) { return '$' + Math.round(c / 100).toLocaleString('es-AR'); }
    function esc(s) { var d = document.createElement('div'); d.textContent = s || ''; return d.innerHTML; }
    function getCode() { try { return sessionStorage.getItem('av_discount') || ''; } catch (e) { return ''; } }
    function setCode(v) { try { sessionStorage.setItem('av_discount', v); } catch (e) {} }
    var busy = false, timer = null;

    function enhance(cart) {
      var inner = CD.querySelector('.drawer__inner');
      if (!inner) return;
      var empty = !cart || cart.item_count === 0;
      var header = CD.querySelector('.drawer__header');
      var h = CD.querySelector('.drawer__heading');
      if (h) h.innerHTML = 'Tu carrito' + (empty ? '' : ' <span class="av-cd-count">' + cart.item_count + '</span>');
      CD.querySelectorAll('.av-cd-banner,.av-cd-up,.av-cd-lbl,.av-cd-disc,.av-cd-disc-msg,.av-cd-save,.av-cd-pay,.av-cd-trust').forEach(function (el) { el.remove(); });
      if (empty || !header) { CD.dataset.avDone = '1'; return; }

      var scroll = CD.querySelector('cart-drawer-items .drawer__contents') || CD.querySelector('cart-drawer-items');
      if (!scroll) { CD.dataset.avDone = '1'; return; }
      var banner = document.createElement('div');
      banner.className = 'av-cd-banner';
      banner.innerHTML = ICON.truck + '<span>Envío gratis a sucursal en la <b>Faja invisible</b>. A domicilio, desde 2 sobres.</span>';
      scroll.insertAdjacentElement('afterbegin', banner);
      var ci = CD.querySelector('cart-drawer-items'); if (ci) ci.scrollLeft = 0;

      var inCart = cart.items.map(function (i) { return String(i.product_id); });
      var tpl = document.getElementById('av-cd-ups');
      var cands = tpl ? Array.prototype.slice.call(tpl.content.querySelectorAll('[data-id]')) : [];
      var fajaId = tpl ? tpl.dataset.faja : '';
      var hasFaja = fajaId && inCart.indexOf(fajaId) !== -1;
      var pick = cands.find(function (c) {
        if (inCart.indexOf(c.dataset.pid) !== -1) return false;
        if (c.dataset.req === 'faja' && !hasFaja) return false;
        return true;
      });
      if (pick) {
        var up = document.createElement('div');
        up.className = 'av-cd-up';
        up.innerHTML = '<p class="av-cd-up__lbl">★ ' + esc(pick.dataset.label || 'Sumalo a tu pedido') + '</p><div class="av-cd-up__row">' +
          (pick.dataset.img ? '<img src="' + pick.dataset.img + '" alt="" width="58" height="58" loading="lazy">' : '') +
          '<div class="av-cd-up__txt"><b>' + esc(pick.dataset.title) + '</b><span>' + esc(pick.dataset.price) + '</span></div>' +
          '<button type="button" class="av-cd-up__btn" data-id="' + pick.dataset.id + '">+ Agregar</button></div>';
        banner.insertAdjacentElement('afterend', up);
      }

      var lbl = document.createElement('p'); lbl.className = 'av-cd-lbl'; lbl.textContent = 'En tu carrito';
      (CD.querySelector('.av-cd-up') || banner).insertAdjacentElement('afterend', lbl);

      var foot = CD.querySelector('.cart-drawer__footer');
      if (foot) {
        var disc = document.createElement('div');
        disc.className = 'av-cd-disc';
        disc.innerHTML = '<input type="text" id="AvCdCode" placeholder="Código de descuento" autocomplete="off" value="' + esc(getCode()) + '"><button type="button">Aplicar</button>';
        var msg = document.createElement('p'); msg.className = 'av-cd-disc-msg';
        if (getCode()) msg.textContent = '✓ El código ' + getCode() + ' se aplica al finalizar la compra.';
        foot.insertAdjacentElement('beforebegin', disc);
        disc.insertAdjacentElement('afterend', msg);
        if (cart.total_discount > 0) {
          var sv = document.createElement('div'); sv.className = 'av-cd-save';
          sv.innerHTML = '<span>✓ Estás ahorrando</span><span>-' + money(cart.total_discount) + '</span>';
          foot.insertAdjacentElement('afterbegin', sv);
        }
        var tl = CD.querySelector('.totals__total'); if (tl) tl.textContent = 'Subtotal';
        var tv = CD.querySelector('.totals__total-value'); if (tv) tv.textContent = money(cart.total_price);
      }
      var btn = CD.querySelector('#CartDrawer-Checkout');
      if (btn) btn.innerHTML = ICON.lock + '<span>Finalizar compra · ' + money(cart.total_price) + '</span>';
      var ctas = CD.querySelector('.cart__ctas');
      if (ctas) {
        var payTpl = document.getElementById('av-cd-pay');
        if (payTpl && payTpl.content.children.length) {
          var pay = document.createElement('ul'); pay.className = 'av-cd-pay';
          pay.appendChild(payTpl.content.cloneNode(true));
          ctas.insertAdjacentElement('afterend', pay);
        }
        var tr = document.createElement('div'); tr.className = 'av-cd-trust';
        tr.innerHTML = '<div>' + ICON.shield + 'Pago seguro</div><div>' + ICON.truck + 'Envíos a todo el país</div><div>' + ICON.card + 'Cuotas con Mercado Pago</div>';
        (CD.querySelector('.av-cd-pay') || ctas).insertAdjacentElement('afterend', tr);
      }
      CD.dataset.avDone = '1';
    }

    function refresh() {
      if (busy) return;
      busy = true;
      fetch('/cart.js', { headers: { Accept: 'application/json' } })
        .then(function (r) { return r.json(); })
        .then(function (cart) { enhance(cart); })
        .catch(function () {})
        .finally(function () { setTimeout(function () { busy = false; }, 50); });
    }
    function needs() {
      var btn = CD.querySelector('#CartDrawer-Checkout');
      var h = CD.querySelector('.drawer__heading');
      return (h && h.textContent.trim().indexOf('Tu carrito') !== 0) || (btn && !btn.querySelector('svg')) || (CD.querySelector('.drawer__header') && !CD.querySelector('.av-cd-banner') && CD.querySelector('cart-drawer-items .cart-item'));
    }
    new MutationObserver(function () {
      clearTimeout(timer);
      timer = setTimeout(function () { if (needs()) refresh(); }, 120);
    }).observe(CD, { childList: true, subtree: true });
    document.addEventListener('cart:av-refresh', refresh);
    refresh();

    /* "Agregar al carrito" desde cualquier parte (ficha, inicio, packs): siempre abre el carrito lateral, nunca la página del carrito */
    document.addEventListener('submit', function (e) {
      var f = e.target;
      if (!f || !f.getAttribute || (f.getAttribute('action') || '').indexOf('/cart/add') === -1) return;
      if (typeof CD.renderContents !== 'function') return;
      e.preventDefault();
      e.stopImmediatePropagation();
      var b = f.querySelector('[type="submit"]') || document.querySelector('[type="submit"][form="' + f.id + '"]');
      var old = f.parentNode && f.parentNode.querySelector('.av-cd-err'); if (old) old.remove();
      if (b) { b.disabled = true; b.style.opacity = '.7'; }
      var fd = new FormData(f);
      fd.append('sections', 'cart-drawer,cart-icon-bubble');
      fd.append('sections_url', window.location.pathname);
      fetch('/cart/add.js', { method: 'POST', headers: { Accept: 'application/javascript', 'X-Requested-With': 'XMLHttpRequest' }, body: fd })
        .then(function (r) { return r.json(); })
        .then(function (res) {
          if (res && res.status) {
            var m = document.createElement('p'); m.className = 'av-cd-err';
            m.textContent = res.description || res.message || 'No se pudo agregar. Probá de nuevo.';
            f.insertAdjacentElement('afterend', m);
            return;
          }
          CD.renderContents(res);
        })
        .catch(function () { f.submit(); })
        .finally(function () { if (b) { b.disabled = false; b.style.opacity = ''; } });
    }, true);

    CD.addEventListener('click', function (e) {
      var add = e.target.closest('.av-cd-up__btn');
      if (add) {
        add.disabled = true; add.textContent = 'Agregando…';
        fetch('/cart/add.js', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json', Accept: 'application/javascript' },
          body: JSON.stringify({ items: [{ id: Number(add.dataset.id), quantity: 1 }], sections: 'cart-drawer,cart-icon-bubble', sections_url: window.location.pathname })
        }).then(function (r) { return r.json(); }).then(function (res) {
          if (res && res.sections && typeof CD.renderContents === 'function') { CD.renderContents(res); }
          else { window.location.reload(); }
        }).catch(function () { add.disabled = false; add.textContent = '+ Agregar'; });
        return;
      }
      var apply = e.target.closest('.av-cd-disc button');
      if (apply) {
        var input = CD.querySelector('#AvCdCode');
        var code = (input && input.value || '').trim().toUpperCase();
        setCode(code);
        var m = CD.querySelector('.av-cd-disc-msg');
        if (m) m.textContent = code ? '✓ El código ' + code + ' se aplica al finalizar la compra.' : '';
        return;
      }
      var co = e.target.closest('#CartDrawer-Checkout');
      if (co) {
        var c = getCode();
        if (c) { e.preventDefault(); window.location.href = '/checkout?discount=' + encodeURIComponent(c); }
      }
    });
  })();
