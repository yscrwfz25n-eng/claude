<script>
  (function () {
    var root = document.querySelector('#MainContent .product__info-container');
    if (!root) return;
    var packs = root.querySelector('[data-av-packs]');
    var bar = document.getElementById('AvSticky-{{ section.id }}');
    function radios() { return root.querySelectorAll('variant-selects input[type=radio]'); }
    function setPrice(txt) { if (bar && txt) bar.querySelector('[data-av-price]').textContent = txt; }

    if (packs && radios().length) {
      root.classList.add('av-pk-ready');
      var checked = root.querySelector('variant-selects input[type=radio]:checked');
      packs.querySelectorAll('.av-pk__card').forEach(function (c) {
        var on = checked && c.dataset.value === checked.value;
        c.classList.toggle('is-on', !!on);
        if (on) setPrice(c.dataset.price);
      });
      packs.addEventListener('click', function (e) {
        var card = e.target.closest('.av-pk__card');
        if (!card || card.disabled) return;
        var radio = Array.prototype.find.call(radios(), function (r) { return r.value === card.dataset.value; });
        if (radio && !radio.checked) radio.click();
        packs.querySelectorAll('.av-pk__card').forEach(function (c) { c.classList.toggle('is-on', c === card); });
        setPrice(card.dataset.price);
      });
    }

    if (bar && 'IntersectionObserver' in window) {
      var target = root.querySelector('.product-form__submit');
      if (!target) return;
      new IntersectionObserver(function (entries) {
        var show = !entries[0].isIntersecting && entries[0].boundingClientRect.top < 0;
        bar.classList.toggle('is-show', show);
        bar.setAttribute('aria-hidden', show ? 'false' : 'true');
        document.body.classList.toggle('av-sticky-on', show);
      }).observe(target);
      bar.querySelector('.av-sticky__btn').addEventListener('click', function () {
        var b = root.querySelector('.product-form__submit');
        if (b && !b.disabled) b.click();
      });
    }
  })();
</script>
