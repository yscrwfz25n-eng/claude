<script>
  (function () {
    var root = document.querySelector('#MainContent .product__info-container');
    var src = document.getElementById('av-src');
    if (!root || !src) return;
    document.body.classList.add('av-faja');
    function top(el) { while (el && el.parentElement !== root) el = el.parentElement; return el; }
    function after(ref, el) { if (ref && el) ref.insertAdjacentElement('afterend', el); }
    var title = top(root.querySelector('.product__title'));
    var buy = top(root.querySelector('product-form'));
    var bul = src.querySelector('[data-av-bul]');
    var wrap = src.querySelector('[data-av-packs-wrap]');
    var packs = src.querySelector('[data-av-packs]');
    var note = src.querySelector('[data-av-note]');
    var acc = src.querySelector('[data-av-acc]');
    after(title, bul);
    if (buy) buy.insertAdjacentElement('beforebegin', wrap);
    after(buy, note);
    after(note, acc);

    function radios() { return root.querySelectorAll('variant-selects input[type=radio]'); }
    function mark() {
      var c = root.querySelector('variant-selects input[type=radio]:checked');
      packs.querySelectorAll('.av-pk__card').forEach(function (k) { k.classList.toggle('is-on', !!c && k.dataset.value === c.value); });
    }
    function pick(value) {
      var r = Array.prototype.find.call(radios(), function (x) { return x.value === value; });
      if (r && !r.checked) r.click();
      packs.querySelectorAll('.av-pk__card').forEach(function (k) { k.classList.toggle('is-on', k.dataset.value === value); });
    }
    if (!radios().length) { wrap.remove(); return; }
    root.classList.add('av-pk-ready');
    mark();
    packs.addEventListener('click', function (e) {
      var card = e.target.closest('.av-pk__card');
      if (card && !card.disabled) pick(card.dataset.value);
    });

    /* 3 sobres elegido de entrada (si no vienen de un link con otro pack) */
    if (!/[?&]variant=/.test(location.search) && window.customElements) {
      customElements.whenDefined('variant-selects').then(function () {
        var rec = packs.querySelector('.av-pk__card[data-n="3"]:not([disabled])');
        if (rec) setTimeout(function () { pick(rec.dataset.value); }, 50);
      });
    }
  })();
</script>
